PARTE A: 
- Implementar o modelo lógico em modelo físico:

CREATE TABLE CONTRIBUINTE (
    CPF INT NOT NULL primary key,
    nome VARCHAR(45) NOT NULL,
    endereco VARCHAR(60),
    telefone VARCHAR(15)
);

CREATE TABLE DECLARACAO_IMPOSTO_RENDA (
    Contribuinte_CPF INT NOT NULL,
    ano_exercicio DATE NOT NULL,
    ano_base DATE NOT NULL,
    valor INT NOT NULL,
    restituicao_ou_pagamento BOOLEAN NOT NULL,

    PRIMARY KEY (Contribuinte_CPF, ano_exercicio),
    FOREIGN KEY (Contribuinte_CPF) REFERENCES CONTRIBUINTE(CPF)
);

CREATE TABLE MALHA_FINA (
    Identificador INT NOT NULL primary key,
    ano_exercicio DATE NOT NULL
);

CREATE TABLE CONTRIBUINTE_MALHA_FINA (
    Malha_Fina_Identificador INT NOT NULL,
    Contribuinte_CPF INT NOT NULL,
    valor INT NOT NULL,
    quantidade_de_prestacoes INT NOT NULL,

    PRIMARY KEY (Malha_Fina_Identificador, Contribuinte_CPF),

    FOREIGN KEY (Malha_Fina_Identificador) REFERENCES MALHA_FINA(Identificador),
    FOREIGN KEY (Contribuinte_CPF) REFERENCES CONTRIBUINTE(CPF)
);
