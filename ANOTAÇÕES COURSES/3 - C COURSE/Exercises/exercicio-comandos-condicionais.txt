3. Faça um programa que leia três números e imprima eles em 
ordem crescente:
#include <stdio.h>

int main() {
    int a, b, c, i, j, temp;

    printf("Digite um numero inteiro: ");
    scanf("%d", &a);
    printf("\nDigite um numero inteiro: ");
    scanf("%d", &b);
    printf("\nDigite um numero inteiro: ");
    scanf("%d", &c);

    int v[3] = {a, b, c};

    for (i = 0; i < 2; i++) {
        for (j = 0; j < 2 - i; j++) {
            if (v[j] > v[j + 1]) {
                temp = v[j];
                v[j] = v[j + 1];
                v[j + 1] = temp;
            }
        }
    }

    for (i = 0; i < 3; i++) {
        printf("%d ", v[i]);
    }

    return 0;
}
